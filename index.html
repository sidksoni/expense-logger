2. Backend Script (One-Time Setup)

  1. Open your Household Expense Google Sheet.
  2. Go to Extensions → Apps Script.
  3. Delete any existing code in Code.gs and paste the script below.
  4. Click Deploy → New deployment.
  5. Select type Web app, set Execute as to Me, and Who has access to Anyone.
  6. Click Deploy, authorize permissions, and paste the Web App URL above.
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    // Targets the first sheet tab (or change to ss.getSheetByName("Sheet1"))
    var sheet = ss.getSheets()[0];
    
    var data = JSON.parse(e.postData.contents);
    
    // Appends row in exact column order:
    // 1. Date | 2. Time | 3. Category | 4. Description | 5. Transaction | 6. Amount | 7. Medium | 8. Account
    sheet.appendRow([
      data.Date,
      data.Time,
      data.Category,
      data.Description,
      data.Transaction,
      Number(data.Amount),
      data.Medium,
      data.Account
    ]);
    
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
